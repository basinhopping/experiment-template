# shell: bash on the node
# expect: "TREE OK {{digest}}", "tests on GPU <one of {{gpus}}>", then the tests' "Ran <n> tests" and "OK", "ENV OK", "check: both passed" and "check.sh exit 0"
# stop if: "a session of {{project}} is running", "TREE DIFFERS", "no free GPU", a test FAILED or ERROR, "ENV DIFFERS", or any "check.sh exit" but 0: stop here and paste the output
cd ~ && mkdir -p ~/{{project}}/logs/{{round}} && { A=$(ls -t ~/{{project}}_incoming/{{project}}_code_*.tar.gz | head -1 | xargs -n1 basename) && if tmux ls -F '#{session_name}' 2>/dev/null | grep -q '^{{project}}_'; then echo "a session of {{project}} is running: stop here and paste this"; tmux ls; else bash ~/{{project}}_incoming/unpack.sh "$A" && D=$(bash ~/{{project}}/node/tree_digest.sh ~/{{project}}) && if [ "$D" = {{digest}} ]; then echo "TREE OK $D"; env -u CUDA_VISIBLE_DEVICES CHECK_GPUS={{gpus}} bash ~/{{project}}/node/check.sh {{environment}} {{profiles}} 2>&1 | grep -E '^(Ran |OK|FAILED|ERROR|ENV |check: |tests on GPU|no free GPU)'; C=${PIPESTATUS[0]}; echo "check.sh exit $C"; if [ "$C" != 0 ]; then tail -n 5 "$(ls -t ~/{{project}}/logs/check_{{environment}}_*.log 2>/dev/null | head -1)" 2>/dev/null; fi; else echo "TREE DIFFERS $D: stop here and paste this"; fi; fi; } 2>&1 | tee -a ~/{{project}}/logs/{{round}}/02_install_test.txt
